SW6 - SEDE SANTIAGO

enable
configure terminal
no ip domain-lookup
hostname SW6
enable secret PRODUCTO
username admin privilege 15 secret PRODUCTO
ip domain-name NEXIUM.COM
crypto key generate rsa modulus 1024
ip ssh version 2
line vty 0 4
 login local
 transport input ssh
exit
line con 0
 login local
exit

! Creación de la base de datos de VLANs
vlan 50
 name CENTRO_DE_DATOS
vlan 60
 name VENTAS
vlan 70
 name ADMINISTRACION
vlan 80
 name SERVIDORES
exit

! IP de Gestión en VLAN 80
interface Vlan80
 description GESTION_SW6
 ip address 10.128.3.146 255.255.255.240
 no shutdown
exit
ip default-gateway 10.128.3.145

! Troncal principal hacia el Router SANTIAGO
interface Ethernet0/0
 description TRONCAL_HACIA_ROUTER
 switchport trunk encapsulation dot1q
 switchport mode trunk
exit

! EtherChannel (LACP) hacia SW9
interface range Ethernet0/1 - 3
 description LACP_HACIA_SW9
 switchport trunk encapsulation dot1q
 switchport mode trunk
 channel-group 1 mode active
exi

enable
configure terminal

! ========================================================
! 1. CONFIGURACIÓN DE SEGURIDAD IPSEC (ISAKMP / IKEv1)
! ========================================================

! Política de Fase 1 (Negociación compatible con imágenes IOL)
crypto isakmp policy 10
 encr aes 256
 hash sha
 authentication pre-share
 group 14
 lifetime 86400
exit
! Contraseña de la VPN compartida actualizada
cr
! Política de Fase 2 (Transform Set para el cifrado de los datos)
crypto ipsec transform-set TS_VPN esp-aes 256 esp-sha-hmac
 mode transport
exit
! Perfil IPSec para amarrar la seguridad al túnel
crypto ipsec profile PERFIL_VPN
 set transform-set TS_VPN
exit

 ========================================================
 2. CONFIGURACIÓN DE LA INTERFAZ TÚNEL (HUB DMVPN)

interface Tunnel0
 description TUNEL DMVPN - HUB CENTRAL
 ip address 10.128.10.1 255.255.255.0
 
tunnel mode gre multipoint
tunnel source Ethernet0/0
 ip nhrp network-id 100
 ip nhrp authentication PRODUCTO3
 ip nhrp map multicast dynamic
 tunnel protection ipsec profile PERFIL_VPN
 ip mtu 1400
 ip tcp adjust-mss 1360
 no shutdown
exit

end
write memory

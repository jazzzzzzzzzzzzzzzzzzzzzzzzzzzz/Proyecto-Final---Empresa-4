SW10

enable
configure terminal
no ip domain-lookup
hostname SW10
enable secret PRODUCTO
username admin privilege 15 secret PRODUCTO
ip domain-name NEXIUM.COM
crypto key generate rsa modulus 1024
ip ssh version 2
line vty 0 4
 login local
 transport input ssh
exit
line con 0
 login local
exit
vlan 120
 name LAN
exit


! Configurar el puerto e0/0 como Troncal hacia el Router
interface Ethernet0/0
 description HACIA ROUTER PUERTO PLATA
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 120
 no shutdown
exit

! Configurar los puertos e0/1 y e0/2 como Troncales hacia SW11
interface range Ethernet0/1-2
 description ENLACES REDUNDANTES HACIA SW11
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 120
 no shutdown
exit

! Interfaz de Gestión
interface Vlan120
 description GESTION
 ip address 10.128.3.162 255.255.255.248
 no shutdown
exit
